# A simple linear circuit - no program built, just the plain work/rest/rounds
# fields plus a station list, exactly as this app has always supported. The
# whole class moves through the same stations together, one per round, and
# each one carries its own instruction for the work phase and the rest phase.
# (For real per-station timing, repeat groups, and more, build a program
# instead - press S, then use the Program tab. See the main README.)
prep=15
work=40
rest=20
rounds=8
restafterfinalround=false
sound=true
volume=0.9
worklabel=WORK
restlabel=REST

# One station per round: Name|during work|during rest|colour, joined with ;
# The colour is optional and shows as a small swatch next to the name.
stations=Exercise bike|80% resistance|no resistance|#2AA7A0;Skip rope|full pace|walk|#D9822B;Rower|max effort|light pull|#5B8DEF;Battle ropes|fast waves|rest arms|#C0453C

# Show each station's own name instead of the plain WORK/REST/MOVE word...
showstationnameasdescription=true
# ...in a table down the left of the screen...
showstationstable=true
# ...with the one in progress highlighted, because this is a linear session
# (everyone on the same station at the same time). Set this to false instead
# for a rotating circuit, where every station is already in use by a
# different group and highlighting just one would be misleading.
highlightcurrentstation=true

# After the rest, take 8 seconds to physically move to the next station
# before the next work phase begins. Skipped after the final round.
usemovetime=true
moveseconds=8
movemessage=Move clockwise to the next station
movecolour=#1E9AA6

presets=45/15x10,40/20x10,30/30x10,20/10x8,60/15x8
